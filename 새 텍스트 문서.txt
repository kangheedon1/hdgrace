3. **`ScriptIntegrator` 클래스: 코드의 영혼**


class ScriptIntegrator:
    def __init__(self):
        self.total_bytes_written = 0
        self.errors_fixed = {"python": 0, "javascript": 0, "html_errors": 0, ...}

    def generate_chrome_command_line_clean(self) -> str:
        chrome_flags = ["--disable-gpu", "--no-sandbox", ...]

        return " ".join(chrome_flags)

    def save_configs(self) -> Tuple[str, str]:
        database_config_str, accounts_config_str, html_code_str 를 파일로 저장


기능JavaScript, HTML, Python 코드를 하나의 XML 파일 내에 **안전하게 통합**하고, Chrome 설정을 **최적화**합니다.

    ChromeCommandLine 정제 (핵심)**:
       `"--disable-features=CookieDeprecationFacilitatedTesting"` **완전 제거** → **BAS 29.2.0의 가장 중요한 보안/호환성 수정**.
       `and "CookieDeprecation" not in flag` → **미래 대비**. Google이 다른 관련 플래그를 추가해도 차단됩니다.
   **설정 저장**:
       `database_config_str`: BAS의 DB 연결 정보를 포함.
       `accounts_config_str`: 5,000개 이상의 계정 정보를 포함 (예: `<아이디>hgildong</아이디><비번>a1b2c3</비번>...`).
       `html_code_str`: 웹 인터페이스의 전체 HTML + CSS + JS 코드.
       오류 통계: `errors_fixed` 사전 → 자동 교정 시스템의 작동을 추적합니다.
   실무적 가치: 시스템의 '생명' 을 결정합니다. 이 클래스가 없으면, BAS 29.2.0 XML은 실행 불가능한 빈 종이가 됩니다.

@app.route('/api/export/ui_elements/<format_type>')
def export_ui_elements_api(format_type):
    hdgrace_system_local = HDGRACECommercialComplete()
    ui_elements = hdgrace_system_local.ui_manager.generate_all_ui_elements()
    export_data = hdgrace_system_local.export_manager.export_ui_elements(ui_elements, format_type)
    if format_type == 'json':
        return jsonify(export_data)
    elif format_type == 'csv':
        csv_content = hdgrace_system_local.export_manager.format_as_csv(export_data)
        response = make_response(csv_content)
        response.headers["Content-Disposition"] = "attachment; filename=ui_elements.csv"
        return response

# hdgrace Repository – Download Links & AI-Folder Archive

GitHub’s REST API supports downloading the **entire** repository as a ZIP or TAR archive. Unfortunately it does *not* support creating ZIPs of individual sub-directories directly via a single API call. Below you’ll find:

1. Links to grab the **entire** `hdgrace` repo as ZIP or TAR  

2. How to list and download the `ai/` folder contents via the Contents API  

3. A sample shell script to automate fetching and zipping only the `ai/` folder

---

## 1. Full-Repo Archives

- ZIP (default branch, e.g. `main` or `master`):
  ```
  https://api.github.com/repos/kangheedon1/hdgrace/zipball
  ```
- TAR (default branch):
  ```
  https://api.github.com/repos/kangheedon1/hdgrace/tarball
  ```
- ZIP (explicit `main` branch via GitHub UI):
  ```
  https://github.com/kangheedon1/hdgrace/archive/refs/heads/main.zip
  ```

These URLs download the **entire** repository, not a sub-directory.

---

## 2. Listing `ai/` Folder via API

To see all files under the `ai/` folder:

```bash
curl -H "Accept: application/vnd.github.v3+json" \
     https://api.github.com/repos/kangheedon1/hdgrace/contents/ai?ref=main
```

Each entry in the JSON response includes a `"download_url"` field—for example:

```json
{
  "name": "model.py",
  "path": "ai/model.py",
  "download_url": "https://raw.githubusercontent.com/kangheedon1/hdgrace/main/ai/model.py",
  ...
}
```

---

## 3. Sample Script: Download & Zip `ai/` Folder

```bash name=download_ai_folder.sh
#!/usr/bin/env bash
# Download all files in hdgrace/ai and create ai_folder.zip

OWNER="kangheedon1"
REPO="hdgrace"
BRANCH="main"
API_URL="https://api.github.com/repos/$OWNER/$REPO/contents/ai?ref=$BRANCH"
TMPDIR=$(mktemp -d)
ZIPFILE="ai_folder.zip"

echo "Listing ai/ folder..."
files=$(curl -s -H "Accept: application/vnd.github.v3+json" "$API_URL" \
         | jq -r '.[] | select(.type=="file") | .download_url')

echo "Downloading ${#files[@]} files…"
for url in $files; do
  filename=$(basename "$url")
  curl -sL "$url" -o "$TMPDIR/$filename"
done

echo "Creating ZIP archive $ZIPFILE…"
(cd "$TMPDIR" && zip -j "../$ZIPFILE" ./*)

echo "Done. Archive located at ./$ZIPFILE"
```

1. **Prerequisites:**  
   - `curl`  
   - `jq` (to parse JSON)  
   - `zip`  

2. **Usage:**  
   ```bash
   chmod +x download_ai_folder.sh
   ./download_ai_folder.sh
   ```

import os

from lxml import etree

# 1) 클론 대상 저장소 리스트
REPO_URLS     = [
    "https://github.com/kangheedon1/hdgracedv2.git",
    "https://github.com/kangheedon1/hdgrace.git"
]
CLONE_DIRS    = [Path(url.split('/')[-1].replace('.git', '')) for url in REPO_URLS]
TREE_OUTPUTS  = [Path(f"repo_tree_{i}.txt") for i in range(len(REPO_URLS))]

# 2) ZIP 대상 디렉토리/파일
SRC_DIRS      = ["ui", "modules", "resources", "configs", "xml"]
EXTRA_FILES   = ["main.py", "README.md", "requirements.txt", "config.yaml"]
ZIP_NAME      = Path("HDGRACE-BAS-Final-AUTO.zip")
ZIP_LIST_FILE = Path("zip_list.txt")

# 3) XML 자동 복구/병합
XML_INPUT_DIR  = Path("extracted/XML")
XML_OUTPUT_DIR = Path("xml/fixed")
MERGED_XML     = Path("HDGRACE-Merged-ALL.xml")

GITHUB_API_URLS = [
    "https://api.github.com/repos/kangheedon1/hdgracedv2/contents",
    "https://api.github.com/repos/kangheedon1/hdgrace/contents"
]
TIMEOUT = 15
logger = logging.getLogger("HDGRACE")

# ------------------------------------------------------------------------------

def clone_repo():
    # 모든 저장소를 depth=1로 클론, 이미 있으면 생략
    for repo_url, clone_dir in zip(REPO_URLS, CLONE_DIRS):
        if not clone_dir.exists():
            print(f"-> Cloning {repo_url} into {clone_dir} (depth=1)...")
            subprocess.run([
                "git", "clone", "--depth=1", repo_url, str(clone_dir)
            ], check=True)
        else:
            print(f"-> Repository already cloned at {clone_dir}")

def dump_repo_tree():
    # 저장소별 전체 파일 트리 기록
    for clone_dir, tree_out in zip(CLONE_DIRS, TREE_OUTPUTS):
        print(f"-> Generating file tree for {clone_dir}...")
        with tree_out.open("w", encoding="utf-8") as out:
            subprocess.run(
                ["git", "ls-tree", "-r", "--long", "HEAD"],
                cwd=str(clone_dir),
                stdout=out,
                check=True
            )
        print(f"   -> Saved file tree to {tree_out}")

def collect_files():
    # 지정 디렉토리/파일에서 중복 없이 모든 파일 고성능 집계
    seen = set()
    files = []
    for d in SRC_DIRS:
        base = Path(d)
        if base.exists():
            for root, _, fnames in os.walk(base):
                for f in fnames:
                    p = Path(root) / f
                    if p not in seen:
                        seen.add(p)
                        files.append(p)
    for fname in EXTRA_FILES:
        p = Path(fname)
        if p.exists() and p not in seen:
            seen.add(p)
            files.append(p)
    return files

def make_zip(files):
    # 모든 파일을 ZIP으로 압축, 내부 목록 zip_list.txt로 기록
    print(f"-> Creating zip archive {ZIP_NAME} with {len(files)} files...")
    with zipfile.ZipFile(ZIP_NAME, "w", zipfile.ZIP_DEFLATED) as zf:
        for p in files:
            zf.write(p, arcname=str(p))
    with zipfile.ZipFile(ZIP_NAME, "r") as zf, ZIP_LIST_FILE.open("w", encoding="utf-8") as lf:
        for info in zf.infolist():
            lf.write(f"{info.filename}\n")
    print(f"   -> ZIP complete: {ZIP_NAME}")
    print(f"   -> Internal file list: {ZIP_LIST_FILE}")

def correct_xml_files():
    # XML 자동 복구(고성능 파서), 누락 없이 모두 복구
    if not XML_INPUT_DIR.exists():
        print(f"-> XML input dir not found: {XML_INPUT_DIR}, skipping XML correction.")
        return
    XML_OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    parser = etree.XMLParser(recover=True, remove_blank_text=True)
    for xml_file in XML_INPUT_DIR.rglob(".xml"):
        rel = xml_file.relative_to(XML_INPUT_DIR)
        out_file = XML_OUTPUT_DIR / rel
        out_file.parent.mkdir(parents=True, exist_ok=True)
        try:
            tree = etree.parse(str(xml_file), parser)
            tree.write(str(out_file), pretty_print=True, encoding="utf-8", xml_declaration=True)
            print(f"   ✔ Corrected XML: {xml_file} -> {out_file}")
        except Exception as e:
            print(f"   ✖ Failed to correct {xml_file}: {e}")

def download_xml(url):
    # GitHub에서 XML 다운로드, 실패 시 빈 문자열 반환(누락 방지)
    try:
        response = requests.get(url, timeout=TIMEOUT)
        response.raise_for_status()
        return response.text
    except requests.exceptions.RequestException as e:
        logger.warning(f"⚠️ 다운로드 실패: {url} - {e}")
        return ""

def fetch_all_github_xml():

    for api_url in GITHUB_API_URLS:
        try:
            resp = requests.get(api_url, timeout=TIMEOUT)
            resp.raise_for_status()
            for item in resp.json():
                name = item.get("name", "").lower()
                url = item.get("download_url")
                if name.endswith(".xml") and url:
                    content = download_xml(url)
                    xmls.append(content)
                    logger.info(f"✔️ 가져옴: {name}")
        except requests.exceptions.RequestException as e:
            logger.warning(f"⚠️ GitHub 접근 실패: {api_url} - {e}")
            xmls.append("")
    return xmls

class UIManager:
    # UI 요소 생성 및 관리 - visible 3중 체크 완전 구현
    def __init__(self):
        self.ui_elements: List[UIElement] = []
        self.id_registry: Set[str] = set()
        self.logger = logging.getLogger(f"{__name__}.UIManager")

    def generate_unique_id(self, prefix: str = "ui") -> str:
        # 중복 방지 고유 ID 생성
        while True:
            unique_id = f"{prefix}_{secrets.token_hex(8)}"
            if unique_id not in self.id_registry:
                self.id_registry.add(unique_id)
                return unique_id

    def create_ui_element(self, element_type: str, index: int) -> UIElement:
        # 개별 UI 요소 생성 - visible 3중 체크 강화
        element_id = self.generate_unique_id(f"hdgrace_{element_type}")

        # 기본 속성 - 3중 체크 필수 적용
        properties = {
            "id": element_id,
            "class": f"hdgrace-{element_type}",
            "data-index": str(index),
            "visible": "true",              # 1. BAS 표준 visible (필수)
            "enabled": "true",
            "style": "hdgrace_style",
            "data-visible": "true",         # 2. HTML5 data-visible (필수)
            "aria-visible": "true"          # 3. 접근성 aria-visible (필수)
        }

        # 타입별 특화 속성
        if element_type == "button":
            properties.update({
                "type": "button",
                "onclick": f"hdgraceButtonClick_{index}()",
                "value": f"HDGRACE Button {index}",
                "role": "button"
            })
        elif element_type == "toggle":
            properties.update({
                "type": "checkbox",
                "onchange": f"hdgraceToggleChange_{index}()",
                "checked": "true", # Changed from " True" to "true" for consistency
                "role": "switch"
            })
        elif element_type == "input":
            properties.update({
                "type": "text",
                "placeholder": f"HDGRACE Input {index}",
                "oninput": f"hdgraceInputChange_{index}()",
                "role": "textbox"
            })

        # 이벤트 바인딩
        events = ["click", "hover", "focus", "blur", "change"]
        if element_type in ["input", "textarea"]:
            events.extend(["input", "keyup", "keydown"])

        # 스타일 속성 - visible 보장 강화
        styles = {
            "display": "block",
            "visibility": "visible",
            "opacity": "1",
            "z-index": "1",
            "position": "relative"
        }

        element = UIElement(
            id=element_id,
            type=element_type,
            name=f"HDGRACE {element_type.title()} {index}",
            visible=True,
            enabled=True,
            properties=properties,
            events=events,
            styles=styles
        )

        return element

    def generate_ui_elements_batch(self, start_index: int, batch_size: int) -> List[UIElement]:
        # UI 요소 배치 생성 (병렬 처리용)
        batch_elements = []

        for i in range(batch_size):
            element_index = start_index + i
            element_type = CommercialConfig.UI_TYPES[element_index % len(CommercialConfig.UI_TYPES)]

            element = self.create_ui_element(element_type, element_index)
            batch_elements.append(element)

        return batch_elements

        with ThreadPoolExecutor(max_workers=CommercialConfig.MAX_WORKERS) as executor:
            futures = []

            # 배치별 병렬 처리

        self.actions: List[Action] = []

        self.total_target_actions = 0
        self.actions_per_ui_distribution = []

    def generate_unique_action_id(self) -> str:
        # 액션 고유 ID 생성
        while True:

                self.action_id_registry.add(action_id)

        # 매크로 고유 ID 생성
        while True:
            macro_id = f"hdgrace_macro_{secrets.token_hex(6)}"
            if macro_id not in self.macro_id_registry:
                self.macro_id_registry.add(macro_id)
                return macro_id

    def calculate_actions_per_ui(self) -> Tuple[int, List[int]]:
        # UI별 액션 분배 계산 - 각 UI당 20~40개 액션
        actions_per_ui = []

        # 각 UI 요소당 랜덤 20~40개 액션 할당
        for i in range(CommercialConfig.TOTAL_UI_ELEMENTS):
            actions_count = random.randint(CommercialConfig.MIN_ACTIONS_PER_MACRO, CommercialConfig.MAX_ACTIONS_PER_MACRO)
            actions_per_ui.append(actions_count)

        total_actions = sum(actions_per_ui)

        # 61,300~122,600 범위 조정
        if total_actions < CommercialConfig.MIN_ACTIONS:
            # 부족한 경우 추가 액션 분배
            additional_needed = CommercialConfig.MIN_ACTIONS - total_actions
            for i in range(additional_needed):
                ui_index = i % CommercialConfig.TOTAL_UI_ELEMENTS
                actions_per_ui[ui_index] += 1
        elif total_actions > CommercialConfig.MAX_ACTIONS:
            # 초과한 경우 액션 감소
            excess = total_actions - CommercialConfig.MAX_ACTIONS
            for i in range(excess):
                ui_index = i % CommercialConfig.TOTAL_UI_ELEMENTS
                if actions_per_ui[ui_index] > CommercialConfig.MIN_ACTIONS_PER_MACRO:
                    actions_per_ui[ui_index] -= 1

        actual_total = sum(actions_per_ui)

        self.logger.info(f"UI별 액션 분배: UI={CommercialConfig.TOTAL_UI_ELEMENTS:,}개, 총액션={actual_total:,}개")

        return actual_total, actions_per_ui

    def safe_cdata_wrap(self, content: str) -> str:
        # CDATA 래핑 - ]]> 이스케이핑 포함 (강화)
        if not content:
            return "<![CDATA[]]>"

        # ]]> 이스케이핑 처리 강화
        escaped_content = str(content).replace(']]>', ']]]]><![CDATA[>')
        return f"<![CDATA[{escaped_content}]]>"

    def normalize_url(self, url: str) -> str:
        # URL 정규화 - 공백 제거 및 처리
        if not url:
            return url

        # 공백 제거
        normalized = url.strip()
        # 여러 공백을 하나로 변환
        normalized = re.sub(r'\s+', '', normalized)
        # URL 인코딩 문제 해결
        normalized = normalized.replace(' ', '%20')

        return normalized

class ExportManager:
    # 데이터 구조 내보내기 관리자 - 다양한 형식 지원
    def __init__(self):
        self.logger = logging.getLogger(f"{__name__}.ExportManager")
        self.supported_formats = ["json", "csv", "xml", "txt"]

    def export_ui_elements(self, ui_elements: List[UIElement], format_type: str = "json") -> Dict[str, Any]:
        # UI 요소 데이터 내보내기
        export_data = []

        for ui in ui_elements:
            ui_dict = {
                "id": ui.id,
                "type": ui.type,
                "name": ui.name,
                "visible": ui.visible,
                "enabled": ui.enabled,
                "properties": ui.properties,
                "events": ui.events,
                "styles": ui.styles
            }
            export_data.append(ui_dict)

        return {
            "format": format_type,
            "type": "ui_elements",
            "count": len(export_data),
            "data": export_data,
            "exported_at": datetime.now().isoformat()
        }

    def export_actions(self, actions: List[Action], format_type: str = "json") -> Dict[str, Any]:
        # 액션 데이터 내보내기
        export_data = []

        for action in actions:
            action_dict = {
                "id": action.id,
                "name": action.name,
                "type": action.type,
                "target": action.target,
                "data": action.data,
                "visible": action.visible,
                "enabled": action.enabled,
                "timeout": action.timeout,
                "retry": action.retry,
                "priority": action.priority
            }
            export_data.append(action_dict)

        return {
            "format": format_type,
            "type": "actions",
            "count": len(export_data),
            "data": export_data,
            "exported_at": datetime.now().isoformat()
        }

    def export_macros(self, macros: List[Macro], format_type: str = "json") -> Dict[str, Any]:
        # 매크로 데이터 내보내기
        export_data = []

        for macro in macros:
            macro_dict = {
                "id": macro.id,
                "name": macro.name,
                "type": macro.type,
                "actions_count": len(macro.actions),
                "action_ids": [action.id for action in macro.actions],
                "conditions": macro.conditions,
                "variables": macro.variables,
                "visible": macro.visible,
                "enabled": macro.enabled
            }
            export_data.append(macro_dict)

        return {
            "format": format_type,
            "type": "macros",
            "count": len(export_data),
            "data": export_data,
            "exported_at": datetime.now().isoformat()
        }

    def export_structure_analysis(self, components: Dict[str, Any], stats: Dict[str, Any], format_type: str = "json") -> Dict[str, Any]:
        # 전체 구조 분석 결과 내보내기
        structure_analysis = {
            "analysis_summary": {
                "ui_elements_count": len(components.get("ui_elements", [])),
                "actions_count": len(components.get("actions", [])),
                "macros_count": len(components.get("macros", [])),
                "api_modules_count": len(components.get("api_modules", [])),
                "analysis_timestamp": datetime.now().isoformat()
            },
            "ui_elements_distribution": {},
            "actions_distribution": {},
            "macro_action_mapping": {},
            "performance_metrics": stats.get("basic_statistics", {}),
            "quality_metrics": stats.get("quality_assurance", {}),
            "compliance_status": stats.get("compliance_summary", {})
        }

        # UI 요소 타입 분포
        ui_elements = components.get("ui_elements", [])
        ui_type_dist = {}
        for ui in ui_elements:
            ui_type_dist[ui.type] = ui_type_dist.get(ui.type, 0) + 1
        structure_analysis["ui_elements_distribution"] = ui_type_dist

        # 액션 타입 분포
        actions = components.get("actions", [])
        action_type_dist = {}
        for action in actions:
            action_type_dist[action.type] = action_type_dist.get(action.type, 0) + 1
        structure_analysis["actions_distribution"] = action_type_dist

        # 매크로-액션 매핑
        macros = components.get("macros", [])
        macro_mapping = {}
        for macro in macros:
            macro_mapping[macro.id] = {
                "name": macro.name,
                "actions_count": len(macro.actions),
                "ui_index": macro.variables.get("ui_index", -1)
            }
        structure_analysis["macro_action_mapping"] = macro_mapping

        return {
            "format": format_type,
            "type": "structure_analysis",
            "data": structure_analysis,
            "exported_at": datetime.now().isoformat()
        }

    def format_as_csv(self, export_data: Dict[str, Any]) -> str:
        # 데이터를 CSV 형식으로 변환
        if export_data["type"] == "ui_elements":
            csv_content = "ID,Type,Name,Visible,Enabled,Properties_Count,Events_Count\n"
            for item in export_data["data"]:
                csv_content += f"{item['id']},{item['type']},{item['name']},{item['visible']},{item['enabled']},{len(item['properties'])},{len(item['events'])}\n"

        elif export_data["type"] == "actions":
            csv_content = "ID,Name,Type,Target,Visible,Enabled,Timeout,Retry,Priority\n"
            for item in export_data["data"]:
                csv_content += f"{item['id']},{item['name']},{item['type']},{item['target']},{item['visible']},{item['enabled']},{item['timeout']},{item['retry']},{item['priority']}\n"

        elif export_data["type"] == "macros":
            csv_content = "ID,Name,Type,Actions_Count,Visible,Enabled\n"
            for item in export_data["data"]:
                csv_content += f"{item['id']},{item['name']},{item['type']},{item['actions_count']},{item['visible']},{item['enabled']}\n"

        elif export_data["type"] == "structure_analysis":
            csv_content = "Analysis Type,HDGRACE Structure Analysis\n"
            csv_content += f"UI Elements Count,{export_data['data']['analysis_summary']['ui_elements_count']}\n"
            csv_content += f"Actions Count,{export_data['data']['analysis_summary']['actions_count']}\n"
            csv_content += f"Macros Count,{export_data['data']['analysis_summary']['macros_count']}\n"
            csv_content += f"Analysis Timestamp,{export_data['data']['analysis_summary']['analysis_timestamp']}\n"

        else:
            csv_content = "Type,Count,Exported_At\n"
            csv_content += f"{export_data['type']},{export_data.get('count', 0)},{export_data['exported_at']}\n"

        return csv_content

    def format_as_xml(self, export_data: Dict[str, Any]) -> str:
        # 데이터를 XML 형식으로 변환
        xml_content = '<?xml version="1.0" encoding="UTF-8"?>\n'
        xml_content += f'<HDGRACEExport type="{export_data["type"]}" exported_at="{export_data["exported_at"]}">\n'

        if export_data["type"] in ["ui_elements", "actions", "macros"]:
            xml_content += f'  <Summary count="{export_data["count"]}" format="{export_data["format"]}"/>\n'
            xml_content += '  <Data>\n'

            for item in export_data["data"][:100]:  # 처음 100개만 (크기 제한)
                xml_content += f'    <Item>\n'
                xml_content += f'      <ID>{item["id"]}</ID>\n'
                xml_content += f'      <Type>{item.get("type", "unknown")}</Type>\n'
                xml_content += f'      <Name><![CDATA[{item.get("name", "")}]]></Name>\n'
                xml_content += f'    </Item>\n'

            xml_content += '  </Data>\n'

        elif export_data["type"] == "structure_analysis":
            data = export_data["data"]
            xml_content += '  <AnalysisSummary>\n'
            for key, value in data["analysis_summary"].items():
                xml_content += f'    <{key}>{value}</{key}>\n'
            xml_content += '  </AnalysisSummary>\n'

        xml_content += '</HDGRACEExport>\n'
        return xml_content

    def save_export_file(self, export_data: Dict[str, Any], output_dir: Path, format_type: str) -> str:
        # 내보내기 파일 저장
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        file_name = f"HDGRACE_{export_data['type']}_{timestamp}.{format_type}"
        file_path = output_dir / file_name

        try:
            if format_type == "json":
                with open(file_path, 'w', encoding='utf-8') as f:
                    json.dump(export_data, f, indent=2, ensure_ascii=False, default=str)

            elif format_type == "csv":
                csv_content = self.format_as_csv(export_data)
                with open(file_path, 'w', newline='', encoding='utf-8') as f:
                    f.write(csv_content)

            elif format_type == "xml":
                xml_content = self.format_as_xml(export_data)
                with open(file_path, 'w', encoding='utf-8') as f:
                    f.write(xml_content)

            elif format_type == "txt":
                with open(file_path, 'w', encoding='utf-8') as f:
                    f.write(f"HDGRACE Export - {export_data['type']}\n")
                    f.write("="  50 + "\n\n")
                    f.write(f"Export Time: {export_data['exported_at']}\n")
                    f.write(f"Format: {export_data['format']}\n")
                    f.write(f"Count: {export_data.get('count', 'N/A')}\n\n")
                    f.write(json.dumps(export_data, indent=2, ensure_ascii=False, default=str))

            self.logger.info(f"내보내기 파일 저장 완료: {file_path}")
            return str(file_path)

        except Exception as e:
            self.logger.error(f"내보내기 파일 저장 실패: {e}")
            raise

    def save_configs(self) -> Tuple[str, str]:
        """Save configurations to files and return paths."""
        # Ensure OUTPUT_DIR exists
        CommercialConfig.OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

        output_txt_path = CommercialConfig.OUTPUT_DIR / "hdgrace_integrated_config.txt"
        output_full_path = CommercialConfig.OUTPUT_DIR / "hdgrace_100_percent_corrected_full.txt"

        with open(output_txt_path, "w", encoding="utf-8") as f:
            # Add Database configuration
            f.write(database_config_str)
            self.total_bytes_written += len(database_config_str.encode('utf-8'))
            self.logger.info(f"Database config saved to {output_txt_path}. Bytes written: {self.total_bytes_written}")

            # Add Accounts configuration
            f.write(accounts_config_str)
            self.total_bytes_written += len(accounts_config_str.encode('utf-8'))
            self.logger.info(f"Accounts config saved. Total bytes written: {self.total_bytes_written}")

            # Add HTML and JavaScript code
            f.write(html_code_str)
            self.total_bytes_written += len(html_code_str.encode('utf-8'))
            self.logger.info(f"HTML and JavaScript code saved. Total bytes written: {self.total_bytes_written}")

        # Save 100% corrected version
        with open(output_full_path, "w", encoding="utf-8") as f:
            f.write(f"""# === 100% 수정본 ===
# 생성 시간: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}
# 파이썬, 자바스크립트, HTML, 파이썬 글꼴 PEP8 코드 100% 수정본. 원본 기능 누락, 파괴될 가능성 0. 리팩토링 완료.

{output_txt_path.read_text(encoding='utf-8')}
""")
            self.logger.info(f"100% 수정본 saved to {output_full_path}. Total bytes written: {self.total_bytes_written}")

        return str(output_txt_path), str(output_full_path)

    def save_script_icon(self) -> None:
        """Attempt to save the script icon from Base64 data."""
        try:
            # Extract icon data from database_config_str
            match = re.search(r'<ScriptIcon>(.?)</ScriptIcon>', database_config_str, re.DOTALL)
            if match:
                icon_data_base64 = match.group(1).strip()
                icon_path = CommercialConfig.OUTPUT_DIR / "script_icon.png"
                with open(icon_path, "wb") as f:
                    f.write(base64.b64decode(icon_data_base64))
                self.logger.info(f"ScriptIcon saved as {icon_path}")
            else:
                self.logger.warning("ScriptIcon Base64 data not found in database_config_str.")
        except Exception as e:
            self.logger.warning(f"Error decoding ScriptIcon: {e}. Please provide complete Base64 data.")

    def activate_script(self, database_config_path: str, accounts_config_path: str) -> None:
        """Activate the script and its configurations."""
        try:
            # Parse database settings from the generated file
            # We need to wrap database_config_path content with a root tag to make it valid XML for ET.parse
            with open(database_config_path, 'r', encoding='utf-8') as f:
                db_content = f.read()

            # Find the start and end of the database_config_str within the file
            db_start_index = db_content.find("<Schema/>")
            db_end_index = db_content.find("<ModelList/>") + len("<ModelList/>")
            
            if db_start_index != -1 and db_end_index != -1:
                db_xml_snippet = db_content[db_start_index:db_end_index]
                # Wrap the snippet in a dummy root for parsing
                db_xml_full = f"<Root>{db_xml_snippet}</Root>"
                database_root = ET.fromstring(db_xml_full)

                script_name_elem = database_root.find("ScriptName")
                engine_version_elem = database_root.find("EngineVersion")

                script_name = script_name_elem.text if script_name_elem is not None else "Unknown Script"
                engine_version = engine_version_elem.text if engine_version_elem is not None else "Unknown Version"

                self.logger.info(f"Activating script: {script_name} (Engine Version: {engine_version})")

                # Enable all modules (as per the ModulesMetaJson)
                modules_meta_elem = database_root.find("ModulesMetaJson")
                if modules_meta_elem is not None and modules_meta_elem.text:
                    modules_meta = json.loads(modules_meta_elem.text)
                    for module, enabled in modules_meta.items():
                        if enabled:
                            self.logger.info(f"Enabling module: {module}")
                else:
                    self.logger.warning("ModulesMetaJson not found or empty in database config.")

                # Parse and activate accounts from the generated file
                # accounts_config_path is actually the same file as database_config_path in this setup
                # We need to extract the accounts_config_str part from it.
                accounts_start_index = db_content.find("<accounts note=")
                accounts_end_index = db_content.find("</accounts>") + len("</accounts>")

                if accounts_start_index != -1 and accounts_end_index != -1:
                    accounts_xml_snippet = db_content[accounts_start_index:accounts_end_index]
                    accounts_root = ET.fromstring(accounts_xml_snippet)
                    for record in accounts_root.findall("record"):
                        account_id_elem = record.find("아이디")
                        status_elem = record.find("상태")
                        proxy_elem = record.find("프록시")

                        account_id = account_id_elem.text if account_id_elem is not None else "N/A"
                        status = status_elem.text if status_elem is not None else "N/A"
                        proxy = proxy_elem.text if proxy_elem is not None else "N/A"
                        self.logger.info(f"Activating account: {account_id} (Status: {status}, Proxy: {proxy})")
                else:
                    self.logger.warning("Accounts configuration not found in the generated file.")

                self.logger.info("Loading configurations into software environment...")
                # Placeholder for actual software API call
                if database_root.find("ConnectionIsRemote") is not None and database_root.find("ConnectionIsRemote").text == "true":
                    self.logger.info("Remote connection enabled. Ensure ConnectionServer and ConnectionPort are configured.")
                if database_root.find("HideBrowsers") is not None and database_root.find("HideBrowsers").text == "true":
                    self.logger.info("Browser hiding enabled.")
                if database_root.find("ShowAdvanced") is not None and database_root.find("ShowAdvanced").text == "true":
                    self.logger.info("Advanced settings UI enabled.")
                if database_root.find("IntegrateScheduler") is not None and database_root.find("IntegrateScheduler").text == "true":
                    self.logger.info("Scheduler integration enabled.")
                if database_root.find("SingleInstance") is not None and database_root.find("SingleInstance").text == "true":
                    self.logger.info("Single instance mode enabled.")
                if database_root.find("CopySilent") is not None and database_root.find("CopySilent").text == "true":
                    self.logger.info("Silent copy mode enabled.")
                if database_root.find("IsEnginesInAppData") is not None and database_root.find("IsEnginesInAppData").text == "true":
                    self.logger.info("Engines stored in AppData.")

                for i in range(1, 10):
                    output_visible_elem = database_root.find(f"OutputVisible{i}")
                    if output_visible_elem is not None and output_visible_elem.text == "1":
                        self.logger.info(f"Output {i} is visible.")

                self.logger.info("Script and accounts activated successfully.")
            else:
                self.logger.error("Database configuration snippet not found in the generated file.")

        except ET.ParseError as e:
            self.logger.error(f"Error parsing XML from generated file: {e}")
        except Exception as e:
            self.logger.error(f"Error activating script: {e}")

    def process_file(self, input_path: str, output_txt_path: Optional[str] = None) -> bool:
        input_file = Path(input_path)
        if not input_file.exists():
            logging.error(f"파일 없음: {input_path}")
            return False

        encoding = self.detect_file_encoding(input_path)
        original_content = input_file.read_text(encoding=encoding, errors='ignore')
        normalized_content = self.normalize_imports(original_content)
        fixed = self.extract_and_fix_mixed_code(normalized_content)

        all_python = "\n\n".join(fixed.get("python", []))
        all_js = "\n\n".join(fixed.get("javascript", []))
        all_html = "\n\n".join(fixed.get("html", []))

        combined_txt = f"""# === 원본 내용 ===
{original_content}

# === 교정된 PYTHON 코드 ===
{all_python}

# === 교정된 JAVASCRIPT 코드 ===
{all_js}

# === 교정된 HTML 코드 ===
{all_html}
"""
        if output_txt_path is None:
            output_txt_path = input_file.parent / f"{input_file.stem}_수정본.txt"

        Path(output_txt_path).write_text(combined_txt, encoding='utf-8')
        logging.info(f"수정본 저장 완료: {output_txt_path}")
        return True

    def detect_file_encoding(self, file_path: str) -> str:
        # Placeholder for chardet, assuming utf-8 for simplicity if not available
        try:
            import chardet
            with open(file_path, 'rb') as f:
                raw = f.read()
            result = chardet.detect(raw)
            return result['encoding'] if result['confidence'] > 0.7 else 'utf-8'
        except ImportError:
            logging.warning("chardet not installed. Defaulting to utf-8 for file encoding detection.")
            return 'utf-8'

        # Placeholder for import normalization
        return content

        # Placeholder for code extraction and fixing
